#include <unistd.h>
#include <stdlib.h>
#include <stdio.h>
#include "get_next_line.h"
#include "ft_libft/libft.h"


char *read_file(int fd, char *buffer, char *stash)
{
	size_t	count;
	char *temp;

	buffer = malloc(BUFFER_SIZE + 1);
	if (!buffer)
		return (NULL);
	count = 1;
	while ((!stash || !ft_strchr(stash, '\n')) && count != 0)
	{
		count = read(fd, buffer, BUFFER_SIZE);
		if (count == -1)
		{
			free (buffer);
			free (stash);
			stash = NULL;
			return (NULL);
		}
		if (count == 0)
			break ;
		if (count > 0)
			buffer[count] = '\0';
		temp = stash;
		stash = gnl_join(stash, buffer);
		free(temp);
	}
	free(buffer);
	return (stash);	
}
char *get_next_line(int fd)
{
	static char *stash;
	char *line;
	char *buffer;

	stash = read_file(fd, buffer, stash);
	if (!stash)
		return (NULL);
	line = getline();	
}

#include <fcntl.h>
int main()
{
	int fd;
	fd = open("text.txt", O_RDWR | O_CREAT, 0777);
	char *str = get_next_line(3);
	printf("%s\n", str);
}
